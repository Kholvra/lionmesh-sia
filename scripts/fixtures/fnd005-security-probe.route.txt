import { catatPenyesuaianUntukSesi, opnameStokUntukSesi, transferStokUntukSesi } from "@/core/inventory";

export async function POST(request: Request) {
  const body: unknown = await request.json();
  if (typeof body !== "object" || body === null || !("operation" in body) || !("payload" in body)) {
    return Response.json({ ok: false }, { status: 400 });
  }
  const result = body.operation === "transfer" ? await transferStokUntukSesi(body.payload)
    : body.operation === "opname" ? await opnameStokUntukSesi(body.payload)
      : body.operation === "adjustment" ? await catatPenyesuaianUntukSesi(body.payload)
        : { ok: false as const, message: "Operasi tidak dikenal." };
  return Response.json(result, { status: result.ok ? 200 : 403 });
}

export async function GET() {
  const { defaultValuationStrategy: strategy } = await import("@/core/inventory");
  return Response.json({
    incoming: strategy.calculateIncoming("0", "0", "0", "900719925474.0993", "0.0001"),
    outgoing: strategy.calculateOutgoing("900719925474.0993", "90071992.55", "0.0001", "900719925474.0993"),
    rounding: strategy.calculateIncoming("0", "0", "0", "1", "0.0050"),
  });
}
